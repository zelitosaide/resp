SELECT FI_FT_ID,
       COUNT(*)      AS ficheiros,
       MIN(FI_NAME)  AS exemplo_nome
  FROM INSIGHT.F_FI_FILE
 WHERE FI_ID > 1600765000
 GROUP BY FI_FT_ID;

---

SELECT F_FI_ID,
       COUNT(*)                                     AS linhas,
       SUM(CASE WHEN CC_DUP = 'N' THEN 1 ELSE 0 END) AS nao_dup
  FROM ODS.STG_CC_CCS_CDR
 WHERE F_FI_ID IN (1600765209, 1600765208, 1600765192,
                   1600765191, 1600765190, 1600765188)
 GROUP BY F_FI_ID;

---

SELECT FI_ID,
       FI_FT_ID,
       TO_CHAR(FI_DATE, 'YYYY-MM-DD HH24:MI:SS')             AS fi_date,
       FI_APPROVED,
       FI_RECORDS,
       TO_CHAR(FI_BACKUP_MOVE_TIME, 'YYYY-MM-DD HH24:MI:SS') AS backup_move,
       FI_PARENT_FI_ID,
       FI_NAME
  FROM INSIGHT.F_FI_FILE
 WHERE FI_ID IN (1600765209, 1600765208, 1600765192,
                 1600765191, 1600765190, 1600765188);
